import requests

def get_prices(coins):
    ids = ','.join(coins)
    url = f"https://api.coingecko.com/api/v3/simple/price?ids={ids}&vs_currencies=usd"
    data = requests.get(url).json()
    for coin in coins:
        print(f"{coin.upper()}: ${data[coin]['usd']}")

get_prices(['bitcoin', 'ethereum', 'solana'])

import requests

def show_portfolio():
    # 取得したい通貨のIDを指定
    coins = 'bitcoin,ethereum,solana'
    url = f"https://api.coingecko.com/api/v3/simple/price?ids={coins}&vs_currencies=jpy"

    res = requests.get(url).json()

    print("--- 現在のレート (JPY) ---")
    print(f"BTC: {res['bitcoin']['jpy']:,}円")
    print(f"ETH: {res['ethereum']['jpy']:,}円")
    print(f"SOL: {res['solana']['jpy']:,}円")

show_portfolio()
