BODY: Writing "Thank you for the meeting..." for the 100th time? Let's automate it. This tool takes your rough notes and turns them into a polite, professional email. 
import openai

def draft_email(notes):
    prompt = f"Write a professional email based on these notes: {notes}"
    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

print(draft_email("Meeting at 2pm, thanks for the data, send report by Friday"))

import openai

openai.api_key = "あなたのAPIキー"

def generate_reply(memo):
    prompt = f"以下のメモを元に、丁寧なビジネスメールを作成して：\n{memo}"
    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

memo = "14時会議感謝、データ確認済み、金曜までに修正版送る"
print(generate_reply(memo))
