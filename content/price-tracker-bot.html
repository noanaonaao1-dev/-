Build a bot that monitors product prices and alerts you via Discord. BODY: Ever missed a 50% discount because you weren't looking? This Python script checks Amazon (or any store) every hour and pings your Discord server when the price drops.   ### Architecture 1. **Scraper**: Pulls the current price from the HTML. 2. **Logic**: Compares it with your "Target Price". 3. **Webhook**: Sends a POST request to Discord.   ### The Code 
import requests
from bs4 import BeautifulSoup

URL = "https://www.example-store.com/item-123"
HEADERS = {"User-Agent": "Mozilla/5.0"}
TARGET_PRICE = 5000

def check_price():
    page = requests.get(URL, headers=HEADERS)
    soup = BeautifulSoup(page.content, 'html.parser')
    price = float(soup.find(id="price").get_text().replace('$', ''))

    if price < TARGET_PRICE:
        send_discord_alert(price)

def send_discord_alert(price):
    webhook_url = "YOUR_DISCORD_WEBHOOK_URL"
    data = {"content": f"🚨 PRICE DROP! Now ${price}. Buy here: {URL}"}
    requests.post(webhook_url, json=data)

check_price()

import requests
from bs4 import BeautifulSoup

URL = "https://www.example-store.com/item-123"
HEADERS = {"User-Agent": "Mozilla/5.0"}
TARGET_PRICE = 5000 # 5000円以下なら通知

def check_price():
    res = requests.get(URL, headers=HEADERS)
    soup = BeautifulSoup(res.content, 'html.parser')
    # IDやクラス名はサイトに合わせて変更してください
    price_text = soup.find(id="priceblock_ourprice").get_text()
    price = int(price_text.replace('¥', '').replace(',', ''))

    if price <= TARGET_PRICE:
        send_discord_alert(price)

def send_discord_alert(price):
    webhook_url = "あなたのDISCORD_WEBHOOK_URL"
    data = {"content": f"🚨 値下げ通知！ 現在 {price}円です！急いで！\n{URL}"}
    requests.post(webhook_url, json=data)

check_price()
